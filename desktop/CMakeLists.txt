# ForgeShape for Windows (`DESKTOP-R1`), and the same editor headless on Linux
# for the automated check. ONE core: every platform-neutral source in
# app/src/main/cpp is compiled unchanged; nothing is forked. See
# docs/DESKTOP_AND_TABLET_PLAN.md.
#
#   cmake -S desktop -B build/desktop -DCMAKE_BUILD_TYPE=Release
#   cmake --build build/desktop --config Release
#
# Needs a C++17 compiler, the Vulkan headers and loader (the Vulkan SDK on
# Windows, libvulkan-dev on Linux) and glslc. On Windows it builds
# ForgeShape.exe; on Linux forgeshape_desktop_check (headless, Mesa lavapipe).
# Both build forgeshape_core_selftests, the host self-test runner.
cmake_minimum_required(VERSION 3.20)
project(ForgeShapeDesktop VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

set(FS_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../app/src/main/cpp")
set(FS_RES "${CMAKE_CURRENT_SOURCE_DIR}/../app/src/main/res")
get_filename_component(FS_CORE "${FS_CORE}" ABSOLUTE)
get_filename_component(FS_RES "${FS_RES}" ABSOLUTE)

find_package(Vulkan REQUIRED)
find_program(FS_GLSLC NAMES glslc glslc.exe
    HINTS "$ENV{VULKAN_SDK}/bin" "$ENV{VULKAN_SDK}/Bin" REQUIRED)

if(MSVC)
    add_compile_options(/utf-8 /W3 /permissive- /EHsc /bigobj)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

# --- shaders, ahead of time, exactly as the Android build compiles them -----
set(FS_SHADER_OUT "${CMAKE_CURRENT_BINARY_DIR}/generated_shaders")
file(MAKE_DIRECTORY "${FS_SHADER_OUT}")
set(FS_SHADER_HEADERS "")
foreach(shader surface.vert surface.frag grid.vert grid.frag gizmo.vert gizmo.frag
                outline_mask.vert outline_mask.frag outline.vert outline.frag ui.vert ui.frag)
    set(_src "${FS_CORE}/shaders/${shader}")
    set(_out "${FS_SHADER_OUT}/${shader}.spv.inc")
    add_custom_command(OUTPUT "${_out}"
        COMMAND "${FS_GLSLC}" -O --target-env=vulkan1.0 -mfmt=c "${_src}" -o "${_out}"
        DEPENDS "${_src}" VERBATIM)
    list(APPEND FS_SHADER_HEADERS "${_out}")
endforeach()
add_custom_target(forgeshape_desktop_shaders DEPENDS ${FS_SHADER_HEADERS})

# --- the icons: the Android vector drawables, embedded as text --------------
# One source for an icon on both platforms. Each file becomes one raw string
# literal (split under MSVC's 16 KB literal limit).
file(GLOB FS_ICON_FILES CONFIGURE_DEPENDS "${FS_RES}/drawable/ic_*.xml")
set(FS_ICONS_CPP "${CMAKE_CURRENT_BINARY_DIR}/desktop_icons.cpp")
set(_icons "// Generated by desktop/CMakeLists.txt from app/src/main/res/drawable.\n")
string(APPEND _icons "#include \"desktop_icons.h\"\nnamespace fsdesktop {\nconst EmbeddedIcon kEmbeddedIcons[] = {\n")
set(_count 0)
foreach(icon ${FS_ICON_FILES})
    get_filename_component(_name "${icon}" NAME_WE)
    file(READ "${icon}" _xml)
    string(APPEND _icons "    {\"${_name}\", R\"fsxml(${_xml})fsxml\"},\n")
    math(EXPR _count "${_count} + 1")
endforeach()
string(APPEND _icons "};\nconst size_t kEmbeddedIconCount = ${_count};\n}  // namespace fsdesktop\n")
file(WRITE "${FS_ICONS_CPP}.tmp" "${_icons}")
configure_file("${FS_ICONS_CPP}.tmp" "${FS_ICONS_CPP}" COPYONLY)

# --- the CAD kernel (GATE-KERNEL), built exactly as the Android build does ---
file(GLOB FS_MANIFOLD_SOURCES CONFIGURE_DEPENDS "${FS_CORE}/third_party/manifold/src/*.cpp")
add_library(forgeshape_manifold STATIC ${FS_MANIFOLD_SOURCES})
target_include_directories(forgeshape_manifold PUBLIC "${FS_CORE}/third_party/manifold/include")
target_compile_definitions(forgeshape_manifold PUBLIC
    MANIFOLD_PAR=-1 MANIFOLD_NO_IOSTREAM MANIFOLD_NO_FILESYSTEM)
if(MSVC)
    target_compile_options(forgeshape_manifold PRIVATE /O2 /w)
else()
    target_compile_options(forgeshape_manifold PRIVATE -O2 -w)
endif()

# --- the one core: every platform-neutral source, unchanged -----------------
file(GLOB FS_CORE_ALL CONFIGURE_DEPENDS "${FS_CORE}/forgeshape_*.cpp")
set(FS_CORE_SOURCES "")
set(FS_SELFTEST_SOURCES "")
foreach(src ${FS_CORE_ALL})
    get_filename_component(_n "${src}" NAME)
    if(_n STREQUAL "forgeshape_jni.cpp")
        continue()
    elseif(_n MATCHES "_selftest\\.cpp$")
        list(APPEND FS_SELFTEST_SOURCES "${src}")
    else()
        list(APPEND FS_CORE_SOURCES "${src}")
    endif()
endforeach()
add_library(forgeshape_core STATIC ${FS_CORE_SOURCES})
add_dependencies(forgeshape_core forgeshape_desktop_shaders)
target_include_directories(forgeshape_core PUBLIC "${FS_CORE}" "${FS_SHADER_OUT}")
target_link_libraries(forgeshape_core PUBLIC forgeshape_manifold Vulkan::Vulkan)

# --- the host self-test runner: every suite, on this compiler ---------------
add_executable(forgeshape_core_selftests
    "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/host_selftest_main.cpp" ${FS_SELFTEST_SOURCES})
target_link_libraries(forgeshape_core_selftests PRIVATE forgeshape_core)

# --- ForgeShape's own interface kit and the desktop editor ------------------
add_library(forgeshape_desktop STATIC
    src/ui/raster.cpp src/ui/truetype.cpp src/ui/vector_icon.cpp src/ui/atlas.cpp
    src/ui/theme.cpp src/ui/ui.cpp src/png_writer.cpp src/desktop_editor.cpp
    "${FS_ICONS_CPP}")
target_include_directories(forgeshape_desktop PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(forgeshape_desktop PUBLIC forgeshape_core)

if(WIN32)
    add_executable(ForgeShape WIN32 src/main_win32.cpp src/forgeshape.rc)
    target_link_libraries(ForgeShape PRIVATE forgeshape_desktop comdlg32 shell32 ole32 dwmapi)
    set_target_properties(ForgeShape PROPERTIES OUTPUT_NAME "ForgeShape")
else()
    add_executable(forgeshape_desktop_check src/main_headless.cpp)
    target_link_libraries(forgeshape_desktop_check PRIVATE forgeshape_desktop)
endif()
